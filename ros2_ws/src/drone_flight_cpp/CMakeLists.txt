cmake_minimum_required(VERSION 3.8)
project(drone_flight_cpp)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(px4_ros2_cpp REQUIRED)
add_executable(square_flight_cpp src/square_flight.cpp)
target_link_libraries(square_flight_cpp Eigen3::Eigen px4_ros2_cpp::px4_ros2_cpp rclcpp::rclcpp)
ament_target_dependencies(square_flight_cpp px4_msgs)
install(TARGETS square_flight_cpp DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
